cmake_minimum_required(VERSION 3.20)
project(throwaway_native_transcription LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_BUILD_TYPE Debug CACHE STRING "Prototype configuration" FORCE)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_IS_DEV OFF CACHE BOOL "" FORCE)
set(WHISPER_COREML OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
set(GGML_METAL ON CACHE BOOL "")
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
if(NOT WHISPER_SOURCE)
    message(FATAL_ERROR "Set WHISPER_SOURCE to pinned whisper.cpp v1.9.5")
endif()
add_subdirectory("${WHISPER_SOURCE}" whisper)
add_executable(whisper_helper whisper_helper.cpp)
target_link_libraries(whisper_helper PRIVATE whisper)
option(PROTOTYPE_CACHED_LIBS "Build CT2/sherpa with approved existing native libraries" ON)
if(PROTOTYPE_CACHED_LIBS)
    include(sherpa.cmake)
    include(ct2.cmake)
endif()
